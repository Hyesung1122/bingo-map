-- ============================================================
-- 1. USERS (안태건) — 기존과 동일, 변경 없음
-- ============================================================
CREATE SEQUENCE users_seq START WITH 1 INCREMENT BY 1 NOCACHE;

CREATE TABLE users (
    id                  NUMBER          NOT NULL,
    name                VARCHAR2(50)    NOT NULL,
    nickname            VARCHAR2(30),
    email               VARCHAR2(100)   NOT NULL,
    password            VARCHAR2(255)   NOT NULL,
    nationality         VARCHAR2(50),
    sns_type            VARCHAR2(20),
    role                VARCHAR2(20)    DEFAULT 'USER' NOT NULL,
    security_question   VARCHAR2(100),
    security_answer     VARCHAR2(255),
    notify_email        VARCHAR2(1),
    location_enabled    VARCHAR2(1),
    created_at          TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at          TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,

    CONSTRAINT pk_users PRIMARY KEY (id),
    CONSTRAINT uq_users_email UNIQUE (email),
    CONSTRAINT ck_users_role CHECK (role IN ('USER', 'ADMIN'))
);

CREATE OR REPLACE TRIGGER users_bi
    BEFORE INSERT ON users
    FOR EACH ROW
BEGIN
    IF :NEW.id IS NULL THEN
        SELECT users_seq.NEXTVAL INTO :NEW.id FROM dual;
    END IF;
END;
/


-- ============================================================
-- 2. RESTAURANTS (장준환) — 기존 + OSM_ID/IS_PUBLISHED 추가
-- (RestaurantDbLoader가 이 두 컬럼을 요구함. 이미 restaurants 테이블이
--  있다면 CREATE 대신 아래 ALTER 두 줄만 실행하면 됨)
-- ============================================================

CREATE SEQUENCE restaurants_seq
    START WITH 1
    INCREMENT BY 1
    NOCACHE;

CREATE TABLE restaurants (
    id                  NUMBER          NOT NULL,
    name                VARCHAR2(100)   NOT NULL,
    category            VARCHAR2(50),
    tags                VARCHAR2(200),
    rating              NUMBER(3,1),
    review_count        NUMBER(10),
    description         VARCHAR2(1000),
    address             VARCHAR2(300),
    latitude            NUMBER(10,7),
    longitude           NUMBER(10,7),
    opening_hours       VARCHAR2(100),
    phone               VARCHAR2(50),
    price_range         VARCHAR2(50),
    website_url         VARCHAR2(300),
    seat_info           VARCHAR2(100),
    reservation_info    VARCHAR2(100),
    payment_methods     VARCHAR2(200),
    languages           VARCHAR2(200),
    main_image_url      VARCHAR2(500),

    menu_name           VARCHAR2(100),
    menu_description    VARCHAR2(500),
    menu_price          VARCHAR2(50),
    menu_image_url      VARCHAR2(500),

    osm_id              NUMBER          UNIQUE,   -- OSM 노드 ID, 수기 등록 식당은 NULL
    is_published        VARCHAR2(1)     DEFAULT 'N' NOT NULL,  -- Y=지도/목록 공개

    created_at          TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at          TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,

    CONSTRAINT pk_restaurants
        PRIMARY KEY (id)
);

-- 이미 restaurants 테이블이 있고 데이터도 있는 경우엔 위 CREATE 대신 이 두 줄만:
-- ALTER TABLE restaurants ADD osm_id NUMBER UNIQUE;
-- ALTER TABLE restaurants ADD is_published VARCHAR2(1) DEFAULT 'N' NOT NULL;

CREATE OR REPLACE TRIGGER restaurants_bi
    BEFORE INSERT ON restaurants
    FOR EACH ROW
BEGIN
    IF :NEW.id IS NULL THEN
        SELECT restaurants_seq.NEXTVAL
        INTO :NEW.id
        FROM dual;
    END IF;
END;
/

-- ============================================================
-- 4. TB_REVIEW / TB_REVIEW_IMAGE (유해성) — 신규
-- 09.16 폴리모픽 설계에서 "식당 리뷰 전용"으로 변경된 최신 구조.
-- 방문일/시간대/목적/추천여부, 이미지 여러 장 지원.
-- rating: Review.java는 Double(0.5 단위 허용) — 0.5 단위 CHECK 유지.
-- recommend_yn: Java Boolean → Oracle NUMBER(1)(0/1)로 가정. 실행해보고
--   Hibernate가 다른 타입을 기대하면(예: CHAR(1)) 여기만 고치면 됨.
-- ============================================================

CREATE SEQUENCE SEQ_REVIEW START WITH 1 INCREMENT BY 1 NOCACHE;

CREATE TABLE TB_REVIEW (
    REVIEW_ID       NUMBER          NOT NULL,
    USER_ID         NUMBER          NOT NULL,
    RESTAURANT_ID   NUMBER          NOT NULL,
    RATING          NUMBER(2,1)     NOT NULL,
    CONTENT         VARCHAR2(2000)  NOT NULL,
    VISIT_DATE      DATE,
    VISIT_TIME_SLOT VARCHAR2(20),
    VISIT_PURPOSE   VARCHAR2(20),
    RECOMMEND_YN    NUMBER(1),
    HELP_COUNT      NUMBER          DEFAULT 0 NOT NULL,
    CREATED_AT      TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,
    CONSTRAINT pk_tb_review PRIMARY KEY (REVIEW_ID),
    CONSTRAINT fk_review_user FOREIGN KEY (USER_ID) REFERENCES users (id),
    CONSTRAINT fk_review_restaurant FOREIGN KEY (RESTAURANT_ID) REFERENCES restaurants (id),
    CONSTRAINT ck_review_rating CHECK (RATING BETWEEN 1 AND 5),
    CONSTRAINT ck_review_rating_half CHECK (MOD(RATING * 2, 1) = 0)
);

CREATE OR REPLACE TRIGGER tb_review_bi
    BEFORE INSERT ON TB_REVIEW
    FOR EACH ROW
BEGIN
    IF :NEW.REVIEW_ID IS NULL THEN
        SELECT SEQ_REVIEW.NEXTVAL INTO :NEW.REVIEW_ID FROM dual;
    END IF;
END;
/

CREATE SEQUENCE SEQ_REVIEW_IMAGE START WITH 1 INCREMENT BY 1 NOCACHE;

CREATE TABLE TB_REVIEW_IMAGE (
    REVIEW_IMAGE_ID NUMBER          NOT NULL,
    REVIEW_ID       NUMBER          NOT NULL,
    IMAGE_URL       VARCHAR2(255)   NOT NULL,
    SORT_ORDER      NUMBER,
    CONSTRAINT pk_tb_review_image PRIMARY KEY (REVIEW_IMAGE_ID),
    CONSTRAINT fk_review_image_review FOREIGN KEY (REVIEW_ID) REFERENCES TB_REVIEW (REVIEW_ID)
);

CREATE OR REPLACE TRIGGER tb_review_image_bi
    BEFORE INSERT ON TB_REVIEW_IMAGE
    FOR EACH ROW
BEGIN
    IF :NEW.REVIEW_IMAGE_ID IS NULL THEN
        SELECT SEQ_REVIEW_IMAGE.NEXTVAL INTO :NEW.REVIEW_IMAGE_ID FROM dual;
    END IF;
END;
/
