Table users {
  id integer [primary key, increment]
  email varchar(100) [not null, unique]
  password varchar(255) [not null]
  name varchar(50) [not null]
  role varchar(20) [not null, default: 'USER', note: 'USER / ADMIN']
  created_at timestamp
  updated_at timestamp
}

Table favorites {
  id integer [primary key, increment]
  user_id integer [not null]
  target_type varchar(20) [not null, note: 'WASTE_BIN / RESTAURANT']
  target_id varchar(100) [not null, note: 'OSM 노드 ID 또는 맛집 PK']
  target_name varchar(200) [note: '목록 표시용 캐시']
  created_at timestamp
  updated_at timestamp

  indexes {
    (user_id, target_type, target_id) [unique]
  }
}

Table reviews {
  id integer [primary key, increment]
  user_id integer [not null]
  target_type varchar(20) [not null]
  target_id varchar(100) [not null]
  rating integer [not null, note: '1~5']
  content varchar(1000)
  created_at timestamp
  updated_at timestamp
}

// ▼ 식당 (변경: restaurants → restaurant, PK id → restaurant_id, OSM_ID / IS_PUBLISHED 추가)
Table restaurant {
  restaurant_id bigint [primary key, increment, note: '시퀀스 SEQ_RESTAURANT']
  name varchar(255) [not null]
  category varchar(255) [note: '타코야키 / 오코노미야키 등']
  tags varchar(255)
  rating decimal(3,1)
  review_count integer
  description varchar(1000)
  address varchar(4000)
  latitude decimal(10,7)
  longitude decimal(10,7)
  opening_hours varchar(1000)
  phone varchar(255)
  price_range varchar(50)
  website_url varchar(1000)
  seat_info varchar(100)
  reservation_info varchar(100)
  payment_methods varchar(200)
  languages varchar(200)
  main_image_url varchar(500) [note: '/images/restaurants/xxx.jpg 형태']
  menu_name varchar(100) [note: '대표 메뉴']
  menu_description varchar(500)
  menu_price varchar(50)
  menu_image_url varchar(500)
  osm_id bigint [unique, note: 'OSM 노드 ID (수기 등록 식당은 NULL)']
  is_published char(1) [not null, default: 'N', note: 'Y = 지도/목록에 공개, N = 비공개']
  created_at timestamp [not null, default: `SYSTIMESTAMP`]
  updated_at timestamp [not null, default: `SYSTIMESTAMP`]
}

Table notices {
  id integer [primary key, increment]
  author_id integer [not null, note: '관리자']
  title varchar(200) [not null]
  content text [not null]
  view_count integer [default: 0]
  created_at timestamp
  updated_at timestamp
}

// ▼ 쓰레기통 (기존 DBML이 이미 팀 SQL과 일치 → 내용 변경 없음)
Table waste_bin {
  id integer [primary key, increment]
  osm_id bigint [not null, unique, note: 'OSM 노드 ID']
  name varchar(100)
  category varchar(20) [not null, note: 'general / recycle / can']
  address varchar(300)
  latitude decimal(10,7) [not null]
  longitude decimal(10,7) [not null]
  city varchar(50) [not null, default: 'osaka']
  created_at timestamp
  updated_at timestamp
}
