-- 기존 객체 정리 (없어도 에러 없이 통과)
BEGIN
    FOR t IN (
        SELECT table_name FROM user_tables
        WHERE table_name IN ('TB_COMMUNITY_COMMENT', 'TB_REVIEW_IMAGE', 'TB_REVIEW', 'TB_COMMUNITY_POST', 'TB_RESTAURANT_MENU',
                             'SYSTEM_SETTINGS', 'REVIEWS', 'FAVORITES', 'NOTICES', 'RESTAURANTS', 'WASTE_BIN', 'USERS')
    ) LOOP
        EXECUTE IMMEDIATE 'DROP TABLE ' || t.table_name || ' CASCADE CONSTRAINTS';
    END LOOP;

    FOR q IN (
        SELECT sequence_name FROM user_sequences
        WHERE sequence_name IN ('USERS_SEQ', 'FAVORITES_SEQ', 'REVIEWS_SEQ',
                                'RESTAURANTS_SEQ', 'NOTICES_SEQ', 'WASTE_BIN_SEQ',
                                'SEQ_REVIEW', 'SEQ_REVIEW_IMAGE', 'SEQ_COMMUNITY_POST',
                                'SEQ_COMMUNITY_COMMENT', 'SEQ_RESTAURANT_MENU')
    ) LOOP
        EXECUTE IMMEDIATE 'DROP SEQUENCE ' || q.sequence_name;
    END LOOP;
END;
/

-- 0918/kdk/팀 쿼리/수정/유저 구분 텍스트 삽입
-- ============================================================
-- USERS
-- 안태건
-- ============================================================
-- 테스트 데이터 id 1~5 고정 → 신규 데이터는 6부터
CREATE SEQUENCE users_seq START WITH 6 INCREMENT BY 1 NOCACHE;

CREATE TABLE users (
    id                  NUMBER          NOT NULL,
    name                VARCHAR2(50)    NOT NULL,
    nickname            VARCHAR2(30),
    email               VARCHAR2(100)   NOT NULL,
    password            VARCHAR2(255)   NOT NULL,
    nationality         VARCHAR2(50),
    sns_type            VARCHAR2(20),
    role                VARCHAR2(20)    DEFAULT 'USER' NOT NULL,
    security_question   VARCHAR2(100),
    security_answer     VARCHAR2(255),
    notify_email        VARCHAR2(1),
    location_enabled    VARCHAR2(1),
    created_at          TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at          TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,

    CONSTRAINT pk_users PRIMARY KEY (id),
    CONSTRAINT uq_users_email UNIQUE (email),
    CONSTRAINT ck_users_role CHECK (role IN ('USER', 'ADMIN'))
);

CREATE OR REPLACE TRIGGER users_bi
    BEFORE INSERT ON users
    FOR EACH ROW
BEGIN
    IF :NEW.id IS NULL THEN
        SELECT users_seq.NEXTVAL INTO :NEW.id FROM dual;
    END IF;
END;
/

-- 0918/kdk/팀 쿼리 수정/레스토랑 파트 구분 텍스트 삽입

-- 테스트 데이터 id 1~5 고정 → 신규 데이터는 6부터
CREATE SEQUENCE restaurants_seq
    START WITH 6
    INCREMENT BY 1
    NOCACHE;

CREATE TABLE restaurants (
    id                  NUMBER          NOT NULL,
    name                VARCHAR2(100)   NOT NULL,
    category            VARCHAR2(50),
    tags                VARCHAR2(200),
    rating              NUMBER(3,1),
    review_count        NUMBER(10),
    description         VARCHAR2(1000),
    address             VARCHAR2(300),
    latitude            NUMBER(10,7),
    longitude           NUMBER(10,7),
    opening_hours       VARCHAR2(100),
    phone               VARCHAR2(50),
    price_range         VARCHAR2(50),
    website_url         VARCHAR2(300),
    seat_info           VARCHAR2(100),
    reservation_info    VARCHAR2(100),
    payment_methods     VARCHAR2(200),
    languages           VARCHAR2(200),
    main_image_url      VARCHAR2(500),

    menu_name           VARCHAR2(100),
    menu_description    VARCHAR2(500),
    menu_price          VARCHAR2(50),
    menu_image_url      VARCHAR2(500),

    created_at          TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at          TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,

    CONSTRAINT pk_restaurants
        PRIMARY KEY (id)
);

CREATE OR REPLACE TRIGGER restaurants_bi
    BEFORE INSERT ON restaurants
    FOR EACH ROW
BEGIN
    IF :NEW.id IS NULL THEN
        SELECT restaurants_seq.NEXTVAL
        INTO :NEW.id
        FROM dual;
    END IF;
END;

-- 0918/kdk/팀 쿼리/수정/리뷰 파트 구분 텍스트 삽입
-- ============================================================
-- REVIEWS
-- 유해성
-- ※ rating: 0.5 단위 별점 지원 (NUMBER(2,1) + 0.5 단위 CHECK)
-- ============================================================

-- 테스트 데이터 id 1~5 고정 → 신규 데이터는 6부터
CREATE SEQUENCE reviews_seq START WITH 6 INCREMENT BY 1 NOCACHE;

CREATE TABLE reviews (
    id          NUMBER          NOT NULL,
    user_id     NUMBER          NOT NULL,
    target_type VARCHAR2(20)    NOT NULL,
    target_id   VARCHAR2(100)   NOT NULL,
    rating      NUMBER(2,1)     NOT NULL,
    content     VARCHAR2(1000),
    created_at  TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at  TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,
    CONSTRAINT pk_reviews PRIMARY KEY (id),
    CONSTRAINT fk_reviews_user FOREIGN KEY (user_id) REFERENCES users (id),
    CONSTRAINT ck_reviews_target_type CHECK (target_type IN ('WASTE_BIN', 'RESTAURANT')),
    CONSTRAINT ck_reviews_rating CHECK (rating BETWEEN 1 AND 5),
    CONSTRAINT ck_reviews_rating_half CHECK (MOD(rating * 2, 1) = 0)
);

CREATE OR REPLACE TRIGGER reviews_bi
    BEFORE INSERT ON reviews
    FOR EACH ROW
BEGIN
    IF :NEW.id IS NULL THEN
        SELECT reviews_seq.NEXTVAL INTO :NEW.id FROM dual;
    END IF;
END;
/

-- ============================================================
-- [v5 추가] TB_REVIEW / TB_REVIEW_IMAGE  (유해성)
-- ※ Review.java가 사용하는 테이블 (방문일/시간대/목적/추천/도움돼요/사진 포함)
-- ※ 사진 파일 자체는 프로젝트의 uploads/reviews 폴더에 저장되고, DB 에는 경로만 저장 (리뷰당 최대 5장)
-- ※ [v7] 자동 번호 트리거 추가
-- ============================================================
CREATE SEQUENCE SEQ_REVIEW START WITH 6 INCREMENT BY 1 NOCACHE;
CREATE SEQUENCE SEQ_REVIEW_IMAGE START WITH 1 INCREMENT BY 1 NOCACHE;

CREATE TABLE TB_REVIEW (
    REVIEW_ID        NUMBER          NOT NULL,                          -- 리뷰 번호 (PK)
    USER_ID          NUMBER          NOT NULL,                          -- 작성자 (users.id)
    RESTAURANT_ID    NUMBER          NOT NULL,                          -- 맛집 번호 (restaurants.id)
    RATING           NUMBER(2,1)     NOT NULL,                          -- 별점 0.5~5
    CONTENT          CLOB            NOT NULL,                          -- 리뷰 내용
    VISIT_DATE       DATE,                                              -- 방문일
    VISIT_TIME_SLOT  VARCHAR2(20),                                      -- 방문 시간대
    VISIT_PURPOSE    VARCHAR2(20),                                      -- 방문 목적 (데이트/여행 등)
    RECOMMEND_YN     NUMBER(1),                                         -- 추천 여부 (1 추천 / 0 비추천)
    HELP_COUNT       NUMBER          DEFAULT 0 NOT NULL,                -- 도움이 돼요 수
    CREATED_AT       TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,     -- 작성 시각
    CONSTRAINT pk_tb_review PRIMARY KEY (REVIEW_ID),
    CONSTRAINT fk_tb_review_user FOREIGN KEY (USER_ID) REFERENCES users (id),
    CONSTRAINT fk_tb_review_restaurant FOREIGN KEY (RESTAURANT_ID) REFERENCES restaurants (id),
    CONSTRAINT ck_tb_review_rating CHECK (RATING BETWEEN 0.5 AND 5)
);

CREATE OR REPLACE TRIGGER tb_review_bi
    BEFORE INSERT ON TB_REVIEW
    FOR EACH ROW
BEGIN
    IF :NEW.REVIEW_ID IS NULL THEN
        SELECT SEQ_REVIEW.NEXTVAL INTO :NEW.REVIEW_ID FROM dual;
    END IF;
END;

-- ============================================================
-- 정합성 점검 (실행 후 아래 기대값과 같은지 확인)
-- ============================================================

-- 기대값: USERS 5 / RESTAURANTS 5 / REVIEWS 5 / TB_REVIEW 5 / NOTICES 5 / TB_COMMUNITY_POST 5
--        TB_COMMUNITY_COMMENT 0 / TB_RESTAURANT_MENU 10 / SYSTEM_SETTINGS 1 / WASTE_BIN 0
SELECT 'USERS' AS tbl, COUNT(*) AS cnt FROM users
UNION ALL SELECT 'RESTAURANTS', COUNT(*) FROM restaurants
UNION ALL SELECT 'REVIEWS', COUNT(*) FROM reviews
UNION ALL SELECT 'TB_REVIEW', COUNT(*) FROM TB_REVIEW
UNION ALL SELECT 'NOTICES', COUNT(*) FROM notices
UNION ALL SELECT 'TB_COMMUNITY_POST', COUNT(*) FROM TB_COMMUNITY_POST
UNION ALL SELECT 'TB_COMMUNITY_COMMENT', COUNT(*) FROM TB_COMMUNITY_COMMENT
UNION ALL SELECT 'TB_RESTAURANT_MENU', COUNT(*) FROM TB_RESTAURANT_MENU
UNION ALL SELECT 'SYSTEM_SETTINGS', COUNT(*) FROM system_settings
UNION ALL SELECT 'WASTE_BIN', COUNT(*) FROM waste_bin;

-- 기대값: USERS_SEQ 6 / RESTAURANTS_SEQ 6 / REVIEWS_SEQ 6 / FAVORITES_SEQ 1 / NOTICES_SEQ 6 / WASTE_BIN_SEQ 1
--        SEQ_REVIEW 6 / SEQ_REVIEW_IMAGE 1 / SEQ_COMMUNITY_POST 6 / SEQ_COMMUNITY_COMMENT 1 / SEQ_RESTAURANT_MENU 11
-- (last_number = 다음에 부여될 번호)
SELECT sequence_name, last_number FROM user_sequences ORDER BY sequence_name;

-- [v7 추가] 기대값: 트리거 11개, 전부 STATUS = ENABLED
--   USERS_BI / RESTAURANTS_BI / FAVORITES_BI / REVIEWS_BI / NOTICES_BI / WASTE_BIN_BI
--   TB_REVIEW_BI / TB_REVIEW_IMAGE_BI / TB_COMMUNITY_POST_BI / TB_COMMUNITY_COMMENT_BI / TB_RESTAURANT_MENU_BI
SELECT trigger_name, table_name, status FROM user_triggers ORDER BY trigger_name;

    id, name, category, tags, rating, review_count,
    description, address, latitude, longitude,
    opening_hours, phone, price_range, website_url,
    seat_info, reservation_info, payment_methods, languages,

-- REVIEWS (작성자+식당+별점+내용, 정합성 확인용)
SELECT
    r.id,
    u.name AS 작성자,
    res.name AS 식당명,
    r.rating AS 별점,
    r.content AS 내용,
    r.created_at AS 작성일
FROM reviews r
JOIN users u
    ON r.user_id = u.id
JOIN restaurants res
    ON r.target_id = TO_CHAR(res.id)
WHERE r.target_type = 'RESTAURANT'
ORDER BY r.id;
