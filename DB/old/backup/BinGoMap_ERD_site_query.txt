Table users {
  id integer [primary key, increment]
  email varchar(100) [not null, unique]
  password varchar(255) [not null]
  name varchar(50) [not null]
  role varchar(20) [not null, default: 'USER', note: 'USER / ADMIN']
  created_at timestamp
  updated_at timestamp
}

Table favorites {
  id integer [primary key, increment]
  user_id integer [not null]
  target_type varchar(20) [not null, note: 'WASTE_BIN / RESTAURANT']
  target_id varchar(100) [not null, note: 'OSM 노드 ID 또는 맛집 PK']
  target_name varchar(200) [note: '목록 표시용 캐시']
  created_at timestamp
  updated_at timestamp

  indexes {
    (user_id, target_type, target_id) [unique]
  }
}

Table reviews {
  id integer [primary key, increment]
  user_id integer [not null]
  target_type varchar(20) [not null]
  target_id varchar(100) [not null]
  rating integer [not null, note: '1~5']
  content varchar(1000)
  created_at timestamp
  updated_at timestamp
}

Table restaurants {
  id integer [primary key, increment]
  name varchar(200) [not null]
  category varchar(50) [note: '타코야키 / 오코노미야키 등']
  address varchar(300)
  latitude decimal [not null]
  longitude decimal [not null]
  image_url varchar(300)
  created_at timestamp
  updated_at timestamp
}

Table notices {
  id integer [primary key, increment]
  author_id integer [not null, note: '관리자']
  title varchar(200) [not null]
  content text [not null]
  view_count integer [default: 0]
  created_at timestamp
  updated_at timestamp
}

Ref: favorites.user_id > users.id
Ref: reviews.user_id > users.id
Ref: notices.author_id > users.id
